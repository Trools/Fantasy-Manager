# World Cup 2026 Fantasy Draft — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a live, NFL-style snake-draft web app for the 2026 World Cup (home use, 4–8 players), hosted entirely on Cloudflare, that decides who drafts which real players.

**Architecture:** One Cloudflare Worker serves a React/Vite/Tailwind SPA (Static Assets), runs a Hono API, and routes WebSockets to a single `DraftRoom` Durable Object that owns live draft state, the pick timer (alarms), and broadcasts. D1 (SQLite) is durable storage. See spec: `docs/superpowers/specs/2026-06-09-worldcup-draft-design.md`.

**Tech Stack:** TypeScript, Cloudflare Workers + Durable Objects + D1 + Static Assets, Hono, React 18 + Vite + Tailwind, Vitest + `@cloudflare/vitest-pool-workers`, Wrangler. Web Crypto (PBKDF2 + HMAC). PDF parsing via `pdfjs-dist` (Node script).

## Parallelization Strategy (agents on different parts)

The build splits into workstreams with this dependency graph:

```
WS0 Foundation & contracts ─────────────► (unblocks everything)
        │
        ├─► WS1 Player data pipeline        (independent)
        ├─► WS2 Pure draft logic            (independent, pure functions)
        ├─► WS3 Auth & crypto backend       (needs schema+types)
        │
        ├─► WS4 DraftRoom Durable Object     (needs WS2)
        ├─► WS5 Hono API routes              (needs WS3)
        │
        └─► WS6 Frontend                     (needs WS0 contracts; visual polish from user's design brief)
                 │
                 └─► WS7 Integration, seed & deploy   (needs all)
```

**Recommended agent waves (Opus subagents, via subagent-driven-development):**

- **Wave A (serial):** WS0 — one agent. Everything depends on the shared types, schema, and test harness. Do not parallelize this.
- **Wave B (3 parallel agents):** WS1, WS2, WS3 — fully independent once WS0 lands. No shared files.
- **Wave C (2 parallel agents):** WS4 (DraftRoom DO, depends on WS2's `draft-logic.ts`) and WS5 (API routes, depends on WS3). Different files; minimal overlap (both touch `db.ts` — WS0 defines it so it's stable).
- **Wave D (parallel per page, 2–5 agents):** WS6 frontend pages. All consume the frozen WS0 protocol/types and the API/WS implemented in C. Each page is a separate file tree. Visual styling slots into shared components the user designs in Claude.
- **Wave E (serial):** WS7 — one agent. Seed, deploy, smoke test.

**Contract-first rule:** WS0 freezes `src/shared/types.ts` (domain types + the full WebSocket message protocol + REST response shapes). Every other workstream codes against it. If a contract must change mid-build, change it in `types.ts` first and notify dependent workstreams.

**Conflict avoidance:** each workstream owns a disjoint set of files (see File Structure). The only shared-write files are `src/shared/types.ts` (frozen in WS0), `wrangler.jsonc` (bindings added in WS0), and `src/worker/index.ts` (entry — WS0 writes it; WS4/WS5 only add route delegations through stable seams defined in WS0). Use isolation worktrees if running Wave B/C/D agents concurrently to avoid working-tree collisions.

```ts
export type DraftStatus = "lobby" | "in_progress" | "paused" | "complete";
export type OrderMode = "snake" | "linear";
export type PosCounts = Record<Position, number>;

export interface Player {
  id: number; country: string; country_code: string; position: Position;
  shirt_number: number | null; full_name: string; name_on_shirt: string | null;
  club: string | null; dob: string | null; active: boolean;
}
export interface PublicUser { id: number; username: string; is_admin: boolean; }
export interface Participant { user_id: number; username: string; draft_order: number | null; joined: boolean; }
export interface Settings {
  total_picks: number; seconds_per_pick: number; pos_min: PosCounts; pos_max: PosCounts;
  max_per_country: number; order_mode: OrderMode;
}
export interface Pick {
  overall_no: number; round_no: number; user_id: number; player_id: number;
  picked_by_user_id: number; picked_at: number;
}
/** Full live snapshot the DO sends on connect and after every change. */
export interface DraftState {
  status: DraftStatus; settings: Settings; participants: Participant[];
  picks: Pick[]; current_pick_no: number | null; current_user_id: number | null;
  timer_deadline: number | null; round_no: number | null;
}

/** Session claims embedded in the signed cookie. */
export interface SessionClaims {
  userId: number; username: string; isAdmin: boolean;
  mustChangePwd: boolean; tokenVersion: number; iat: number;
}

/** ---- WebSocket protocol ---- */
export type ClientMsg =
  | { t: "pick"; player_id: number }
  | { t: "ping" };
export type ServerMsg =
  | { t: "state"; state: DraftState }            // full snapshot
  | { t: "pick_made"; pick: Pick; player: Player; by_username: string }
  | { t: "timer"; deadline: number | null }
  | { t: "error"; code: string; message: string }
  | { t: "pong" };

/** REST response shapes. */
export interface MeResponse { user: PublicUser | null; must_change_password: boolean; }
export interface ErrorResponse { error: string; }

export const POSITIONS: Position[] = ["GK", "DEF", "MID", "FWD"];
```

```ts
export async function getUserByUsername(db: D1Database, username: string): Promise<UserRow | null> {
  return db.prepare("SELECT * FROM users WHERE username = ?").bind(username).first<UserRow>();
}
export async function getUserById(db: D1Database, id: number): Promise<UserRow | null> {
  return db.prepare("SELECT * FROM users WHERE id = ?").bind(id).first<UserRow>();

export async function countUsers(db: D1Database): Promise<number> {
  const r = await db.prepare("SELECT COUNT(*) AS n FROM users").first<{ n: number }>();
  return r?.n ?? 0;
}
export async function getSettingsRow(db: D1Database) {
  return db.prepare("SELECT * FROM draft_settings WHERE id = 1").first<any>();
}
export function parseSettings(row: any): Settings & { status: DraftStatus; current_pick_no: number | null; timer_deadline: number | null } {
  return {
    total_picks: row.total_picks, seconds_per_pick: row.seconds_per_pick,

    status: row.status, current_pick_no: row.current_pick_no, timer_deadline: row.timer_deadline,
  };
}
export async function listParticipants(db: D1Database): Promise<Participant[]> {
  const { results } = await db.prepare(
    `SELECT p.user_id, u.username, p.draft_order, p.joined
     FROM participants p JOIN users u ON u.id = p.user_id ORDER BY p.draft_order, u.username`
  ).all<any>();
  return results.map(r => ({ user_id: r.user_id, username: r.username, draft_order: r.draft_order, joined: !!r.joined }));
}
export async function listPicks(db: D1Database): Promise<Pick[]> {
  const { results } = await db.prepare("SELECT * FROM picks ORDER BY overall_no").all<Pick>();
  return results;
}
export async function getActivePlayers(db: D1Database): Promise<Player[]> {
  const { results } = await db.prepare("SELECT * FROM players WHERE active = 1 ORDER BY country, position").all<any>();
  return results.map(r => ({ ...r, active: !!r.active })) as Player[];
}
```

The PDF (`docs/players/SquadLists-English.pdf`) has 48 team pages. Each page has a numbered list `N POS LASTNAME Firstname` and detail rows giving full name, club `(CC)`, DOB, height. We extract per page: country (from the header line `Country (CODE)`), and for each player number 1–26: position (DF→DEF, MF→MID, FW→FWD, GK→GK), name, and club. Strategy: use `pdfjs-dist` to get text items with x/y coordinates per page; cluster by row (y), then by column (x) to de-interleave the two columns; correlate the numbered POS list with the detail rows by surname.

```js
export function parseCountryHeader(line) {

  return m ? { country: m[1].trim(), country_code: m[2] } : null;

// extractClubCode("Lille OSC (FRA)") -> "Lille OSC" + club country ignored for now
export function cleanClub(s) { return s ? s.replace(/\s*\([A-Z]{3}\)\s*$/, "").trim() : null; }

  for (let p = 1; p <= doc.numPages; p++) {
```

Generate multi-row `INSERT` SQL in chunks of 100 and pipe through wrangler:

```js
import { readFileSync, writeFileSync } from "node:fs";
import { execSync } from "node:child_process";
const remote = process.argv.includes("--remote");
const players = JSON.parse(readFileSync("data/players.json", "utf8"));
const esc = s => s == null ? "NULL" : `'${String(s).replace(/'/g, "''")}'`;
const chunks = [];
for (let i = 0; i < players.length; i += 100) chunks.push(players.slice(i, i + 100));
let sql = "DELETE FROM players;\n";
for (const c of chunks) {
  sql += "INSERT INTO players (country,country_code,position,shirt_number,full_name,name_on_shirt,club,dob) VALUES\n";
  sql += c.map(p => `(${esc(p.country)},${esc(p.country_code)},${esc(p.position)},${p.shirt_number ?? "NULL"},${esc(p.full_name)},${esc(p.name_on_shirt)},${esc(p.club)},${esc(p.dob)})`).join(",\n") + ";\n";
}
writeFileSync(".seed.sql", sql);
execSync(`wrangler d1 execute DRAFT_DB ${remote ? "--remote" : "--local"} --file=.seed.sql`, { stdio: "inherit" });
```

```ts
import type { OrderMode, Position, PosCounts, Player, Pick } from "./types";

export function roundForPickNo(pickNo: number, n: number): number {
  return Math.floor((pickNo - 1) / n) + 1;
}
export function pickerForPickNo(pickNo: number, order: number[], mode: OrderMode): number {
  const n = order.length;
  const round = roundForPickNo(pickNo, n);
  const idxInRound = (pickNo - 1) % n; // 0-based
  const forward = mode === "linear" || round % 2 === 1;
  const idx = forward ? idxInRound : n - 1 - idxInRound;
  return order[idx]!;
}
```

```ts
export interface RosterEntry { position: Position; country_code: string; }
export interface EligResult { ok: boolean; reason?: string; }

export function eligibility(
  pos: Position, countryCode: string, roster: RosterEntry[],
  s: { total_picks: number; pos_min: PosCounts; pos_max: PosCounts; max_per_country: number }
): EligResult {
  const counts: PosCounts = { GK: 0, DEF: 0, MID: 0, FWD: 0 };
  let fromCountry = 0;
  for (const e of roster) { counts[e.position]++; if (e.country_code === countryCode) fromCountry++; }
  if (counts[pos] >= s.pos_max[pos]) return { ok: false, reason: `${pos} full` };
  if (fromCountry >= s.max_per_country) return { ok: false, reason: `Max from ${countryCode}` };
  // forward feasibility: after taking this pick, can remaining picks satisfy all unmet minimums?
  const after: PosCounts = { ...counts, [pos]: counts[pos] + 1 };
  const remainingAfter = s.total_picks - (roster.length + 1);
  let unmet = 0;
  for (const p of ["GK", "DEF", "MID", "FWD"] as Position[]) unmet += Math.max(0, s.pos_min[p] - after[p]);
  if (unmet > remainingAfter) return { ok: false, reason: "would leave minimums unreachable" };
  return { ok: true };
}

/** Convenience: roster derived from picks for a user. */
export function rosterFromPicks(picks: Pick[], players: Map<number, Player>, userId: number): RosterEntry[] {
  return picks.filter(p => p.user_id === userId).map(p => {
    const pl = players.get(p.player_id)!;
    return { position: pl.position, country_code: pl.country_code };
  });
}
```

```ts
import type { SessionClaims } from "./types";
const enc = new TextEncoder();
const ITER = 200_000;

function b64u(buf: ArrayBuffer | Uint8Array): string {
  const b = buf instanceof Uint8Array ? buf : new Uint8Array(buf);
  return btoa(String.fromCharCode(...b)).replace(/\+/g,"-").replace(/\//g,"_").replace(/=+$/,"");
}
function unb64u(s: string): Uint8Array {
  s = s.replace(/-/g,"+").replace(/_/g,"/"); return Uint8Array.from(atob(s), c => c.charCodeAt(0));
}
async function pbkdf2(pw: string, salt: Uint8Array): Promise<string> {
  const key = await crypto.subtle.importKey("raw", enc.encode(pw), "PBKDF2", false, ["deriveBits"]);
  const bits = await crypto.subtle.deriveBits({ name:"PBKDF2", hash:"SHA-256", salt, iterations: ITER }, key, 256);
  return b64u(bits);
}
export async function hashPassword(pw: string): Promise<{ hash: string; salt: string }> {
  const salt = crypto.getRandomValues(new Uint8Array(16));
  return { hash: await pbkdf2(pw, salt), salt: b64u(salt) };
}
export async function verifyPassword(pw: string, hash: string, salt: string): Promise<boolean> {
  const h = await pbkdf2(pw, unb64u(salt));
  // constant-time-ish compare
  if (h.length !== hash.length) return false;
  let diff = 0; for (let i=0;i<h.length;i++) diff |= h.charCodeAt(i) ^ hash.charCodeAt(i);
  return diff === 0;
}
async function hmac(data: string, secret: string): Promise<string> {
  const key = await crypto.subtle.importKey("raw", enc.encode(secret), { name:"HMAC", hash:"SHA-256" }, false, ["sign"]);
  return b64u(await crypto.subtle.sign("HMAC", key, enc.encode(data)));
}
export async function signSession(claims: SessionClaims, secret: string): Promise<string> {
  const body = b64u(enc.encode(JSON.stringify(claims)));
  return `${body}.${await hmac(body, secret)}`;
}
export async function verifySession(token: string, secret: string): Promise<SessionClaims | null> {
  const [body, sig] = token.split(".");
  if (!body || !sig) return null;
  if (await hmac(body, secret) !== sig) return null;
  try { return JSON.parse(new TextDecoder().decode(unb64u(body))) as SessionClaims; } catch { return null; }
}
```

```ts
async webSocketMessage(ws: WebSocket, raw: string) {
  const msg = JSON.parse(raw);
  if (msg.t === "ping") return ws.send(JSON.stringify({ t: "pong" }));
  if (msg.t !== "pick") return;
  const att = ws.deserializeAttachment() as { userId: number };
  // Load current snapshot from in-memory cache (kept fresh on each pick) or rebuild once.
  const state = this.cache ?? (this.cache = await this.buildState());
  const players = this.playerMap ?? (this.playerMap = new Map((await getActivePlayers(this.env.DRAFT_DB)).map(p=>[p.id,p])));
  // ---- SYNCHRONOUS GUARD + MUTATE (no await until after broadcast) ----
  if (state.status !== "in_progress") return ws.send(JSON.stringify({ t:"error", code:"not_active", message:"Draft not active" }));
  if (state.current_user_id !== att.userId) return ws.send(JSON.stringify({ t:"error", code:"not_your_turn", message:"Not your turn" }));
  const player = players.get(msg.player_id);
  if (!player) return ws.send(JSON.stringify({ t:"error", code:"no_player", message:"Unknown player" }));
  if (state.picks.some(p => p.player_id === player.id)) return ws.send(JSON.stringify({ t:"error", code:"taken", message:"Already drafted" }));
  const roster = rosterFromPicks(state.picks, players, att.userId);
  const elig = eligibility(player.position, player.country_code, roster, state.settings);
  if (!elig.ok) return ws.send(JSON.stringify({ t:"error", code:"ineligible", message: elig.reason! }));
  const order = state.participants.filter(p=>p.draft_order!=null).sort((a,b)=>a.draft_order!-b.draft_order!).map(p=>p.user_id);
  const overall = state.current_pick_no!;
  const pick = { overall_no: overall, round_no: roundForPickNo(overall, order.length), user_id: att.userId,
    player_id: player.id, picked_by_user_id: att.userId, picked_at: Date.now() };
  // mutate cache + decide next
  state.picks.push(pick);
  const complete = isDraftComplete(state.picks.length, order, state.settings.total_picks);
  const nextNo = complete ? overall : overall + 1;
  state.current_pick_no = complete ? null : nextNo;
  state.status = complete ? "complete" : "in_progress";
  state.current_user_id = complete ? null : pickerForPickNo(nextNo, order, state.settings.order_mode);
  const deadline = complete ? null : Date.now() + state.settings.seconds_per_pick * 1000;
  state.timer_deadline = deadline;
  this.broadcast({ t:"pick_made", pick, player, by_username: att.username ?? "" } as any);
  this.broadcast({ t:"state", state });
  // ---- AWAIT PERSIST (after state is committed) ----
  await this.persistPick(pick, state.status, state.current_pick_no, deadline);
  if (deadline) await this.ctx.storage.setAlarm(deadline); else await this.ctx.storage.deleteAlarm();
}

private async persistPick(pick: any, status: string, currentNo: number | null, deadline: number | null) {
  await this.env.DRAFT_DB.batch([
    this.env.DRAFT_DB.prepare("INSERT INTO picks (overall_no,round_no,user_id,player_id,picked_by_user_id,picked_at) VALUES (?,?,?,?,?,?)")
      .bind(pick.overall_no, pick.round_no, pick.user_id, pick.player_id, pick.picked_by_user_id, pick.picked_at),
    this.env.DRAFT_DB.prepare("UPDATE draft_settings SET status=?, current_pick_no=?, timer_deadline=? WHERE id=1")
      .bind(status, currentNo, deadline),
  ]);
}
```
